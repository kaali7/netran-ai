01 // Native OS rate integer decimation (48000 / 16000 = 3x boxcar)
02 const factor = Math.round(this._decimationFactor);
03 const outputLength = Math.floor(inputChannel.length / factor);
04 const downsampled = new Float32Array(outputLength);
05
06 for (let i = 0; i < outputLength; i++) {
07 let sum = 0;
08 for (let j = 0; j < factor; j++) sum += inputChannel[i * factor + j];
09 downsampled[i] = sum / factor; // Clean box-car average
10 }
11
12 // Convert Float32 [-1.0, +1.0] to Int16 signed PCM
13 const pcm16 = new Int16Array(downsampled.length);
14 for (let i = 0; i < downsampled.length; i++) {
15 let s = Math.max(-1, Math.min(1, downsampled[i]));
16 pcm16[i] = s < 0 ? s * 0x8000 : s * 0x7FFF;
17 }
18 // Zero-copy transfer ArrayBuffer to main WebSocket thread
19 this.port.postMessage(pcm16.buffer, [pcm16.buffer]);
